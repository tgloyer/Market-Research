
# ==========================================
# DYNAMIC STOP RESEARCH — STEP 1
# PREDICTED RETURN vs MAE
# ==========================================


# ==========================================
# GET AFTERNOON DATA
# ==========================================

afternoon = minute_bars[
    (minute_bars["time"] >= pd.to_datetime("11:30").time()) &
    (minute_bars["time"] < pd.to_datetime("16:00").time())
].copy()


# ==========================================
# GET ENTRY PRICE
# ==========================================

entry_prices = (
    afternoon[
        afternoon["time"] == pd.to_datetime("11:30").time()
    ]
    .groupby("date")["open"]
    .first()
)

results["entry_price"] = entry_prices


# ==========================================
# GET AFTERNOON HIGH / LOW
# ==========================================

afternoon_high = (
    afternoon
    .groupby("date")["high"]
    .max()
)

afternoon_low = (
    afternoon
    .groupby("date")["low"]
    .min()
)


results["afternoon_high"] = afternoon_high
results["afternoon_low"] = afternoon_low


# ==========================================
# CALCULATE MFE
# ==========================================

results["mfe"] = np.where(
    results["position"] == 1,

    (results["afternoon_high"] /
     results["entry_price"]) - 1,

    (results["entry_price"] /
     results["afternoon_low"]) - 1
)


# ==========================================
# CALCULATE MAE
# ==========================================

results["mae"] = np.where(
    results["position"] == 1,

    (results["afternoon_low"] /
     results["entry_price"]) - 1,

    (results["entry_price"] /
     results["afternoon_high"]) - 1
)


# ==========================================
# CREATE PREDICTION BUCKETS
# ==========================================

results["prediction_bucket"] = pd.cut(

    results["predicted_return"],

    bins=[
        -np.inf,
        -0.0030,
        -0.0015,
        0,
        0.0015,
        0.0030,
        np.inf
    ],

    labels=[
        "< -0.30%",
        "-0.30% to -0.15%",
        "-0.15% to 0%",
        "0% to +0.15%",
        "+0.15% to +0.30%",
        "> +0.30%"
    ]
)


# ==========================================
# OVERALL ANALYSIS
# ==========================================

print("\n==========================================")
print("PREDICTED RETURN vs MAE")
print("==========================================")


for bucket, group in results.groupby(
    "prediction_bucket",
    observed=False
):

    print(f"\nPrediction: {bucket}")

    print(
        f"Trades: {len(group)}"
    )

    print(
        f"Win rate: "
        f"{(group['target_return'] > 0).mean():.2%}"
    )

    print(
        f"Average MAE: "
        f"{group['mae'].mean():.2%}"
    )

    print(
        f"Median MAE: "
        f"{group['mae'].median():.2%}"
    )

    print(
        f"75th percentile MAE: "
        f"{group['mae'].quantile(0.75):.2%}"
    )

    print(
        f"90th percentile MAE: "
        f"{group['mae'].quantile(0.90):.2%}"
    )

    print(
        f"Average MFE: "
        f"{group['mfe'].mean():.2%}"
    )

    print(
        f"Average final return: "
        f"{group['target_return'].mean():.2%}"
    )


# ==========================================
# WINNING TRADES
# ==========================================

print("\n==========================================")
print("WINNING TRADES — MAE BY PREDICTION")
print("==========================================")


winning_trades = results[
    results["target_return"] > 0
]


for bucket, group in winning_trades.groupby(
    "prediction_bucket",
    observed=False
):

    print(f"\nPrediction: {bucket}")

    print(
        f"Winning trades: {len(group)}"

    print(

    )

    print(

    )

    print(

        f"{group['mae'].quantile(0.90):.2%}"
    )

    print(
        f"Worst MAE: "
        f"{group['mae'].min():.2%}"
    )

print("\n==========================================")
